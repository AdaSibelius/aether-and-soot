- In `src/sim/spatial/block-geometry.ts:55–64`, `mapPoint` implements 2D coordinate translation with axis flipping:
  ```ts
  export function mapPoint(p: BuildingPlacement, pt: Point2D): Point2D {
    const w = p.src.x2 - p.src.x1;
    const h = p.src.y2 - p.src.y1;
    const u = pt.x - p.src.x1;
    const v = pt.y - p.src.y1;
    return {
      x: p.dst.x1 + (p.mirrorX ? w - u : u),
      y: p.dst.y1 + (p.flipY ? h - v : v),
    };
  }
  ```

## 2. Logic Chain

1. **Declarative CDDA Matrix Representation**:
   - Cataclysm: Dark Days Ahead uses ASCII row matrices where each glyph maps to terrain, furniture, loot tables, and room zones.
   - By creating `src/sim/spatial/building-templates.ts` with a shared legend table (`COMMON_LEGEND`) and 6 distinct building matrices, we eliminate redundant boilerplate and ensure the entire module stays well under the 220-line target (~185 lines).

2. **Discrete Matrix Transformation Invariant (Rotation & Mirroring)**:
   - For a template of local dimensions $W \times H$, local coordinates are $(u, v) \in [0, W - 1] \times [0, H - 1]$.
   - Clockwise rotation transformations:
     * $0^\circ$: $(x_r, y_r) = (u, v)$, with dimensions $(W_r, H_r) = (W, H)$.
     * $90^\circ$: $(x_r, y_r) = (H - 1 - v, u)$, with dimensions $(W_r, H_r) = (H, W)$.
     * $180^\circ$: $(x_r, y_r) = (W - 1 - u, H - 1 - v)$, with dimensions $(W_r, H_r) = (W, H)$.
     * $270^\circ$: $(x_r, y_r) = (v, W - 1 - u)$, with dimensions $(W_r, H_r) = (H, W)$.
   - Mirroring applied post-rotation on $(x_r, y_r)$:
     * `mirrorX`: $x_m = W_r - 1 - x_r$; otherwise $x_m = x_r$.
     * `flipY`: $y_m = H_r - 1 - y_r$; otherwise $y_m = y_r$.
   - Final translation to origin $(O_x, O_y)$:
     * $X = O_x + x_m$, $Y = O_y + y_m$.
   - This formulation guarantees that every coordinate is bijective, bounds-safe, and deterministic across any rotation and reflection combination.

3. **Door Alignment & Outward Normal**:
   - Templates author exterior entrance doors on perimeter walls.
   - The outward normal vector $(\Delta u, \Delta v)$ indicates the direction toward the street.
   - When rotated and mirrored, the outward vector transforms via the identical matrix, ensuring that a door placed on the street side connects directly to the exterior thoroughfare.

4. **100% Interior BFS Walkability Invariant**:
   - Each template specifies discrete room bounding boxes (`rooms: TemplateRoomDefinition[]`).
   - After stamping, `instantiateTemplateIntoGrid` calculates the transformed room box for every room, finds the first walkable floor cell inside that room, and executes `hasWalkablePath(nextGrid, mainDoor, roomWalkablePoint)`.
   - If any interior room is cut off by misaligned walls or blocking furniture, an invariant violation error is triggered, preventing broken floorplans from entering the world state.

5. **Modularity Enforcement (< 220 lines)**:
   - `building-templates.ts`: uses shared legend composition to keep template definitions compact (~185 lines, target <= 220, limit 300).
   - `template-instantiator.ts`: single-responsibility stamping and transform module (~175 lines, target <= 220, limit 300).

export type TemplateRotation = 0 | 90 | 180 | 270;

export interface TemplatePlacement {
  readonly origin: Point2D;
  readonly rotation?: TemplateRotation;
  readonly mirrorX?: boolean;
  readonly flipY?: boolean;
  readonly targetStratum?: StratumId;
}

export interface TemplateInstantiationResult {
  readonly grid: StratumGrid;
  readonly entities: Record<string, EntityRecord>;
  readonly entrancePoints: readonly Point2D[];
  readonly outwardPoints: readonly Point2D[];
  readonly fixturesPlaced: readonly EntityId[];
  readonly doors: Record<string, Point2D>;
  readonly keyPositions: Record<string, Point2D>;
}

export function getTransformedDimensions(
  w: number,
  h: number,
  rotation: TemplateRotation = 0,
): { width: number; height: number } {
  return rotation === 90 || rotation === 270
    ? { width: h, height: w }
    : { width: w, height: h };
}

export function transformCoord(
  u: number,
  v: number,
  w: number,
  h: number,
  rotation: TemplateRotation = 0,
  mirrorX = false,
  flipY = false,
  origin: Point2D = { x: 0, y: 0 },
): Point2D {
  let rx = u;
  let ry = v;
  let rw = w;
  let rh = h;

  switch (rotation) {
    case 90:
      rx = h - 1 - v;
      ry = u;
      rw = h;
      rh = w;
      break;
    case 180:
      rx = w - 1 - u;
      ry = h - 1 - v;
      rw = w;
      rh = h;
      break;
    case 270:
      rx = v;
      ry = w - 1 - u;
      rw = h;
      rh = w;
      break;
    case 0:
    default:
      break;
  }

  const mx = mirrorX ? rw - 1 - rx : rx;
  const my = flipY ? rh - 1 - ry : ry;

  return { x: origin.x + mx, y: origin.y + my };
}

export function transformRoomBox(
  room: TemplateRoomDefinition,
  w: number,
  h: number,
  rotation: TemplateRotation = 0,
  mirrorX = false,
  flipY = false,
  origin: Point2D = { x: 0, y: 0 },
): { x1: number; y1: number; x2: number; y2: number } {
  const c1 = transformCoord(room.x1, room.y1, w, h, rotation, mirrorX, flipY, origin);
  const c2 = transformCoord(room.x2, room.y2, w, h, rotation, mirrorX, flipY, origin);
  return {
    x1: Math.min(c1.x, c2.x),
    y1: Math.min(c1.y, c2.y),
    x2: Math.max(c1.x, c2.x),
    y2: Math.max(c1.y, c2.y),
  };
}

/**
 * Deterministically stamps a declarative BuildingTemplate into a StratumGrid:
 * 1. Applies 4-directional rotation (0/90/180/270) and mirroring (mirrorX/flipY)
 * 2. Stamps terrain, interiorZone, and atmospheric parameters into grid cells
 * 3. Spawns interactive fixtures and portals with deterministic EntityIds
 * 4. Verifies 100% BFS walkability from entrance door to all interior rooms
 */
export function instantiateTemplateIntoGrid(
  grid: StratumGrid,
  entities: Record<string, EntityRecord>,
  template: BuildingTemplate,
  placement: TemplatePlacement,
  rng: SeededRNG,
): TemplateInstantiationResult {
  const rot = placement.rotation ?? 0;
  const mx = placement.mirrorX ?? false;
  const fy = placement.flipY ?? false;
  const origin = placement.origin;
  const stratum = placement.targetStratum ?? grid.stratum;

  const w = template.width;
  const h = template.height;
  const dims = getTransformedDimensions(w, h, rot);

  if (origin.x < 0 || origin.y < 0 || origin.x + dims.width > grid.width || origin.y + dims.height > grid.height) {
    throw new Error(`Template '${template.id}' out of bounds at (${origin.x}, ${origin.y}) on ${grid.width}x${grid.height}`);
  }

  const cells = [...grid.cells];
  const outEntities: Record<string, EntityRecord> = { ...entities };
  const entrancePoints: Point2D[] = [];
  const outwardPoints: Point2D[] = [];
  const fixturesPlaced: EntityId[] = [];
  const doors: Record<string, Point2D> = {};
  const keyPositions: Record<string, Point2D> = {};

  for (let v = 0; v < h; v++) {
    const row = template.rows[v];
    if (!row || row.length !== w) throw new Error(`Template '${template.id}' row ${v} invalid length`);
    for (let u = 0; u < w; u++) {
      const ch = row[u]!;
      const rule = template.legend[ch];
      if (!rule) throw new Error(`Template '${template.id}' missing legend for '${ch}' at (${u}, ${v})`);

      const pt = transformCoord(u, v, w, h, rot, mx, fy, origin);
      const idx = pt.y * grid.width + pt.x;
      const prev = cells[idx]!;

      const blocksMove = rule.blocksMovement ?? (rule.terrain === 'SOOT_BRICK_WALL' || rule.terrain === 'WOODEN_DOOR_CLOSED' || rule.terrain === 'COUNTER_BARRIER' || rule.terrain === 'GLASS_WINDOW' || rule.terrain === 'WROUGHT_IRON_FENCE');
      const blocksSight = rule.blocksSight ?? (rule.terrain === 'SOOT_BRICK_WALL' || rule.terrain === 'WOODEN_DOOR_CLOSED');

      cells[idx] = {
        ...prev,
        terrainType: rule.terrain,
        blocksMovement: blocksMove,
        blocksSight: blocksSight,
        interiorZone: rule.interiorZone ?? prev.interiorZone ?? 'NONE',
        steamPressureKpa: rule.steamPressureKpa ?? prev.steamPressureKpa,
        smokePpm: rule.smokePpm ?? prev.smokePpm,
      };

      if (ch === template.exteriorDoorChar) {
        entrancePoints.push(pt);
        doors[`${template.id}_door_${entrancePoints.length}`] = pt;
        const outU = u === 0 ? -1 : u === w - 1 ? 1 : 0;
        const outV = v === 0 ? -1 : v === h - 1 ? 1 : 0;
        outwardPoints.push(transformCoord(u + outU, v + outV, w, h, rot, mx, fy, origin));
      }

      if (rule.furnitureKind) {
        const fixtureId = EntityIdSchema.parse(`entity_fixture_${rule.furnitureKind.toLowerCase()}_${pt.x}_${pt.y}_${stratum}`);
        outEntities[fixtureId] = createFixtureEntity(rule.furnitureKind, fixtureId, { x: pt.x, y: pt.y, stratum }, rule, rng);
        fixturesPlaced.push(fixtureId);
        keyPositions[`${rule.furnitureKind.toLowerCase()}_${fixturesPlaced.length}`] = pt;
      }

      if (rule.crossStratumPortal) {
        const portalId = EntityIdSchema.parse(`entity_portal_${template.id}_${pt.x}_${pt.y}_${stratum}`);
        outEntities[portalId] = {
          id: portalId,
          name: rule.crossStratumPortal.portalLabel,
          visual: { glyph: 'H', fgPrimary: '#D4AF37' },
          position: { x: pt.x, y: pt.y, stratum, blocksMovement: false, blocksSight: false },
          portal: { target: { x: pt.x, y: pt.y, stratum: rule.crossStratumPortal.targetStratum }, bidirectional: true, conductsGas: true },
        };
      }
    }
  }

  const nextGrid: StratumGrid = { ...grid, cells };

  if (entrancePoints.length > 0) {
    const mainDoor = entrancePoints[0]!;
    for (const room of template.rooms) {
      const box = transformRoomBox(room, w, h, rot, mx, fy, origin);
      let roomWalkable: Point2D | undefined;
      for (let ry = box.y1; ry <= box.y2; ry++) {
        for (let rx = box.x1; rx <= box.x2; rx++) {
          const c = cells[ry * grid.width + rx];
          if (c && !c.blocksMovement) { roomWalkable = { x: rx, y: ry }; break; }
        }
        if (roomWalkable) break;
      }
      if (roomWalkable && !hasWalkablePath(nextGrid, mainDoor, roomWalkable)) {
        throw new Error(`Walkability invariant failed: Room '${room.label}' unreachable from door in '${template.id}'`);
      }
    }
  }

  return { grid: nextGrid, entities: outEntities, entrancePoints, outwardPoints, fixturesPlaced, doors, keyPositions };
}

## 4. Caveats

1. **Boundary Collision with Yard Setbacks**:
   - Building templates have variable bounding boxes (11x10, 12x10, 13x10, 11x11). In Milestone 2 (`urban-district.ts`), plot zoning must allocate parcels with at least a 1-tile setback around the perimeter to ensure exterior doors do not spawn directly on the world grid boundary ($x=0$ or $y=0$).
2. **Door Outward Placement on Rotated Grids**:
   - When rotated by 90° or 270°, the exterior door normal rotates into $x$-facing vectors $(\pm 1, 0)$. Street carving algorithms connecting parcels to thoroughfares in Milestone 2 must trace from `result.outwardPoints[0]` rather than assuming a fixed $y$-axis corridor.
3. **Cellar Undercroft Alignment**:
   - `TEMPLATE_VICTORIAN_TAVERN` and `TEMPLATE_TENEMENT_HOUSE` include `crossStratumPortal` entries targeting `UNDERCROFT`. In Milestone 2, the undercroft procedural generator must ensure that the corresponding $(x, y)$ coordinate on `UNDERCROFT` is a walkable corridor or cistern alcove, not an uncarved solid granite wall.
